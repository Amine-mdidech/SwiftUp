# Firebase Setup + Auth Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add Firebase-backed authentication (Sign in with Apple + email/password) to SwiftUp, gating the app ahead of the existing onboarding flow, with a bare Firestore `users/{uid}` document created on first sign-in.

**Architecture:** A new `AuthRepository` (`@Observable`, matching the existing repository pattern) wraps Firebase Auth and Firestore. `RootView` gets a new outer gate: unauthenticated → `AuthView`; authenticated → the existing `hasCompletedOnboarding` gate, unchanged. Firebase initializes once in `SwiftUpApp.init()`.

**Tech Stack:** SwiftUI, Swift Testing (not XCTest), Firebase iOS SDK via Swift Package Manager (`FirebaseAuth`, `FirebaseFirestore`, `FirebaseCore`), XcodeGen for project generation.

**Spec:** `docs/superpowers/specs/2026-09-08-firebase-auth-setup-design.md`

## Global Constraints

- iOS deployment target: 17.0 (from `project.yml`)
- Tests use **Swift Testing** (`import Testing`, `@Test`, `#expect`), never XCTest — matches every existing test file in `SwiftUpTests/`
- **`SwiftUp.xcodeproj/` is gitignored** (`.gitignore` has `*.xcodeproj/`) — `project.yml` is the only committed source of truth for project configuration. Any Xcode-UI signing/capability change must be mirrored into `project.yml` or it's lost on the next `xcodegen generate`
- Bundle ID: `com.swiftup.dailyquiz`
- Apple Developer Team ID: `5K3Q8T2G5A` (already selected in this project's local Xcode signing; confirmed via the already-working "Sign in with Apple" capability)
- Firebase via Swift Package Manager only — no CocoaPods
- Existing repositories (`ProfileRepository`, `QuestionRepository`, etc.) are plain `@Observable final class`, no protocols — `AuthRepository` follows the same shape (a single concrete class with a `isPreview` constructor flag for tests/previews, rather than introducing a protocol+fake pair, since this codebase has no existing protocol-based repository pattern to match)
- Design system: `Theme.Colors`, `Theme.Fonts`, `Theme.Spacing` (`SwiftUp/DesignSystem/Theme.swift`) and atoms `RuledMasthead`, `EyebrowLabel`, `SegmentedPicker` (`SwiftUp/DesignSystem/Atoms.swift`, `InteractiveAtoms.swift`) — new UI must reuse these, not introduce new styling primitives

import Foundation

enum AuthValidation {
    /// A pragmatic (not RFC 5322-exhaustive) email shape check: something@something.something.
    static func isValidEmail(_ email: String) -> Bool {
        let pattern = #"^[^\s@]+@[^\s@]+\.[^\s@]+$"#
        return email.range(of: pattern, options: .regularExpression) != nil
    }

    /// Matches Firebase Auth's own minimum password length.
    static func isValidPassword(_ password: String) -> Bool {
        password.count >= 6
    }
}

Create `SwiftUp/Data/AuthRepository.swift`:

```swift
import Foundation
import FirebaseAuth
import FirebaseFirestore
import AuthenticationServices
import CryptoKit

@Observable
final class AuthRepository: NSObject {
    private(set) var isSignedIn: Bool
    private(set) var currentUser: AuthUser?
    var errorMessage: String?

    private let isPreview: Bool
    private var authStateHandle: AuthStateDidChangeListenerHandle?

    init(isPreview: Bool = false, signedIn: Bool = false) {
        self.isPreview = isPreview
        self.isSignedIn = isPreview ? signedIn : false
        self.currentUser = (isPreview && signedIn)
            ? AuthUser(uid: "preview-user", email: "preview@example.com", displayName: "Preview User")
            : nil
        super.init()

        guard !isPreview else { return }
        authStateHandle = Auth.auth().addStateDidChangeListener { [weak self] _, user in
            guard let self else { return }
            self.isSignedIn = user != nil
            self.currentUser = user.map {
                AuthUser(uid: $0.uid, email: $0.email, displayName: $0.displayName)
            }
        }
    }

    deinit {
        if let authStateHandle {
            Auth.auth().removeStateDidChangeListener(authStateHandle)
        }
    }

    // MARK: - Email/password

    func signUp(email: String, password: String) async {
        errorMessage = nil
        do {
            let result = try await Auth.auth().createUser(withEmail: email, password: password)
            await ensureUserDocument(uid: result.user.uid, email: result.user.email, displayName: result.user.displayName)
        } catch {
            errorMessage = Self.friendlyMessage(for: error)
        }
    }

    func signIn(email: String, password: String) async {
        errorMessage = nil
        do {
            _ = try await Auth.auth().signIn(withEmail: email, password: password)
        } catch {
            errorMessage = Self.friendlyMessage(for: error)
        }
    }

    func signOut() {
        do {
            try Auth.auth().signOut()
        } catch {
            errorMessage = Self.friendlyMessage(for: error)
        }
    }

    // MARK: - Sign in with Apple

    func handleAppleSignIn(result: Result<ASAuthorization, Error>, rawNonce: String) async {
        errorMessage = nil
        switch result {
        case .failure(let error):
            if (error as? ASAuthorizationError)?.code == .canceled { return }
            errorMessage = "Sign in with Apple failed. Please try again."

        case .success(let authorization):
            guard
                let credential = authorization.credential as? ASAuthorizationAppleIDCredential,
                let tokenData = credential.identityToken,
                let idTokenString = String(data: tokenData, encoding: .utf8)
            else {
                errorMessage = "Sign in with Apple failed. Please try again."
                return
            }

            let firebaseCredential = OAuthProvider.appleCredential(
                withIDToken: idTokenString,
                rawNonce: rawNonce,
                fullName: credential.fullName
            )

            do {
                let result = try await Auth.auth().signIn(with: firebaseCredential)
                let nameFromApple = [credential.fullName?.givenName, credential.fullName?.familyName]
                    .compactMap { $0 }
                    .joined(separator: " ")
                let displayName = result.user.displayName ?? (nameFromApple.isEmpty ? nil : nameFromApple)
                await ensureUserDocument(uid: result.user.uid, email: result.user.email, displayName: displayName)
            } catch {
                errorMessage = Self.friendlyMessage(for: error)
            }
        }
    }

    // MARK: - Firestore

    private func ensureUserDocument(uid: String, email: String?, displayName: String?) async {
        let doc = Firestore.firestore().collection("users").document(uid)
        do {
            let snapshot = try await doc.getDocument()
            guard !snapshot.exists else { return }
            var data: [String: Any] = [
                "uid": uid,
                "createdAt": FieldValue.serverTimestamp()
            ]
            data["email"] = email
            data["displayName"] = displayName
            try await doc.setData(data)
        } catch {
            // Non-blocking: isSignedIn reflects Firebase Auth state only (spec §6).
            // The doc gets created lazily the next time this runs, e.g. next launch.
        }
    }

    // MARK: - Nonce (Apple Sign In replay protection)

    static func randomNonceString(length: Int = 32) -> String {
        precondition(length > 0)
        var randomBytes = [UInt8](repeating: 0, count: length)
        let status = SecRandomCopyBytes(kSecRandomDefault, randomBytes.count, &randomBytes)
        precondition(status == errSecSuccess, "Unable to generate nonce: \(status)")

        let charset: [Character] = Array("0123456789ABCDEFGHIJKLMNOPQRSTUVXYZabcdefghijklmnopqrstuvwxyz-._")
        return String(randomBytes.map { charset[Int($0) % charset.count] })
    }

    static func sha256(_ input: String) -> String {
        let hashed = SHA256.hash(data: Data(input.utf8))
        return hashed.compactMap { String(format: "%02x", $0) }.joined()
    }

    // MARK: - Errors

    private static func friendlyMessage(for error: Error) -> String {
        let nsError = error as NSError
        switch AuthErrorCode(rawValue: nsError.code) {
        case .wrongPassword, .invalidCredential:
            return "Incorrect email or password."
        case .emailAlreadyInUse:
            return "An account already exists with that email."
        case .invalidEmail:
            return "Enter a valid email address."
        case .networkError:
            return "No connection. Check your network and try again."
        case .weakPassword:
            return "Password must be at least 6 characters."
        default:
            return "Something went wrong. Please try again."
        }
    }
}
```

        }
        .background(Theme.Colors.bg)
    }

    private func submit() {
        validationMessage = nil
        guard AuthValidation.isValidEmail(email) else {
            validationMessage = "Enter a valid email address."
            return
        }
        guard AuthValidation.isValidPassword(password) else {
            validationMessage = "Password must be at least 6 characters."
            return
        }
        isSubmitting = true
        Task {
            switch mode {
            case .signIn: await authRepository.signIn(email: email, password: password)
            case .signUp: await authRepository.signUp(email: email, password: password)
            }
            isSubmitting = false
        }
    }
}
